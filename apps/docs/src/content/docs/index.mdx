import { Card, CardGrid } from "@astrojs/starlight/components";

<CardGrid stagger>
  <Card title="Yours, on Cloudflare" icon="rocket">
    The API, dashboard and background jobs run as one Worker in your account. Data lives in your
    D1 database. No third-party service sees your secrets.
  </Card>
  <Card title="Encrypted per project" icon="seti:lock">
    AES-256-GCM per value under a per-project key, wrapped by a root key held as a Worker secret.
    Keys rotate in the background.
  </Card>
  <Card title="Built for CI" icon="seti:shell">
    `anchor run -- <cmd>` injects secrets into a process. Scoped, expiring service tokens keep each
    pipeline to the environments and folders it needs.
  </Card>
  <Card title="Audited" icon="document">
    Every read and change is recorded (never the value), with roles, environment-scoped members and
    a 365-day append-only log.
  </Card>
</CardGrid>
